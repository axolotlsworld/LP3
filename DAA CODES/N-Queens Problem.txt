#include <iostream>
using namespace std;

#define N 4

// Print chessboard
void printBoard(int board[N][N])
{
    for (int i = 0; i < N; i++)
    {
        for (int j = 0; j < N; j++)
        {
            cout << board[i][j] << " ";
        }
        cout << endl;
    }

    cout << endl;
}

// Check whether queen can be placed
bool isSafe(int board[N][N], int row, int col)
{
    // Check same column
    for (int i = 0; i < row; i++)
    {
        if (board[i][col] == 1)
            return false;
    }

    // Check upper left diagonal
    for (int i = row - 1, j = col - 1;
         i >= 0 && j >= 0; i--, j--)
    {
        if (board[i][j] == 1)
            return false;
    }

    // Check upper right diagonal
    for (int i = row - 1, j = col + 1;
         i >= 0 && j < N; i--, j++)
    {
        if (board[i][j] == 1)
            return false;
    }

    return true;
}

// Solve N Queens using backtracking
bool solveNQueens(int board[N][N], int row)
{
    if (row == N)
    {
        printBoard(board);
        return true;
    }

    bool found = false;

    for (int col = 0; col < N; col++)
    {
        if (isSafe(board, row, col))
        {
            board[row][col] = 1;

            if (solveNQueens(board, row + 1))
                found = true;

            // Backtracking
            board[row][col] = 0;
        }
    }

    return found;
}

int main()
{
    int board[N][N] = {0};

    cout << "N-Queens Solutions:\n\n";

    if (!solveNQueens(board, 0))
    {
        cout << "No solution exists";
    }

    return 0;
}
